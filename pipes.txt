#include <iostream>
#include <unistd.h>
#include <sys/wait.h>
#include <fcntl.h>
#include <fstream>
#include <cstring>

using namespace std;

int main()
{
    int pipe1[2];
    int pipe2[2];

    char filename[100];
    char contents[1000];

    // Create two pipes
    pipe(pipe1);
    pipe(pipe2);

    pid_t pid = fork();

    if(pid < 0)
    {
        cout << "Fork failed!" << endl;
        return 1;
    }

    // CHILD PROCESS
    else if(pid == 0)
    {
        close(pipe1[1]);
        close(pipe2[0]);

        // Read filename from parent
        read(pipe1[0], filename, sizeof(filename));

        cout << "Child received filename: " << filename << endl;

        // Open file
        ifstream file(filename);

        if(!file)
        {
            strcpy(contents, "File not found!");
        }
        else
        {
            file.read(contents, sizeof(contents) - 1);
            contents[file.gcount()] = '\0';

            file.close();
        }

        // Send file contents to parent
        write(pipe2[1], contents, strlen(contents) + 1);

        close(pipe1[0]);
        close(pipe2[1]);

        return 0;
    }

    // PARENT PROCESS
    else
    {
        close(pipe1[0]);
        close(pipe2[1]);

        cout << "Enter filename: ";
        cin >> filename;

        // Send filename to child
        write(pipe1[1], filename, strlen(filename) + 1);

        // Read file contents from child
        read(pipe2[0], contents, sizeof(contents));

        cout << "\nFile contents:\n";
        cout << contents << endl;

        close(pipe1[1]);
        close(pipe2[0]);

        wait(NULL);
    }

    return 0;
}
